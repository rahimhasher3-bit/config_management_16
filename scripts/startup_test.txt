import argparse
import os
import shlex
from pathlib import Path


DEFAULT_VFS_NAME = "my_vfs"


def parse_command(user_input):
    expanded_input = os.path.expandvars(user_input)

    try:
        return shlex.split(expanded_input)
    except ValueError as error:
        print(f"Ошибка разбора команды: {error}")
        return None


def command_ls(args):
    print("ls", *args)


def command_cd(args):
    print("cd", *args)


def execute_command(parts):
    if not parts:
        return "continue"

    command = parts[0]
    args = parts[1:]

    if command == "ls":
        command_ls(args)
        return "continue"

    if command == "cd":
        command_cd(args)
        return "continue"

    if command == "exit":
        return "exit"

    print(f"Ошибка: неизвестная команда '{command}'")
    return "error"


def get_vfs_name(vfs_path):
    if not vfs_path:
        return DEFAULT_VFS_NAME

    return Path(vfs_path).stem


def run_startup_script(script_path, vfs_name):
    try:
        with open(script_path, encoding="utf-8") as script_file:
            lines = script_file.readlines()
    except OSError as error:
        print(f"Ошибка чтения стартового скрипта: {error}")
        return "error"

    for line in lines:
        command_line = line.strip()

        if not command_line:
            continue

        print(f"{vfs_name}:~$ {command_line}")

        parts = parse_command(command_line)

        if parts is None:
            print("Стартовый скрипт остановлен.")
            return "error"

        result = execute_command(parts)

        if result == "error":
            print("Стартовый скрипт остановлен.")
            return "error"

        if result == "exit":
            return "exit"

    return "continue"


def parse_arguments():
    parser = argparse.ArgumentParser(
        description="Эмулятор оболочки ОС"
    )

    parser.add_argument(
        "--vfs",
        dest="vfs_path",
        help="Путь к физическому расположению VFS"
    )

    parser.add_argument(
        "--script",
        dest="script_path",
        help="Путь к стартовому скрипту"
    )

    return parser.parse_args()


def print_configuration(args):
    print("Параметры запуска:")
    print(f"VFS: {args.vfs_path or 'не задан'}")
    print(
        f"Стартовый скрипт: "
        f"{args.script_path or 'не задан'}"
    )


def run_repl(vfs_name):
    print("Эмулятор оболочки ОС")
    print("Для выхода введите: exit")

    while True:
        try:
            user_input = input(f"{vfs_name}:~$ ")
            parts = parse_command(user_input)

            if parts is None:
                continue

            result = execute_command(parts)

            if result == "exit":
                break

        except KeyboardInterrupt:
            print("\nДля выхода используйте команду exit.")

        except EOFError:
            print()
            break


def main():
    args = parse_arguments()
    vfs_name = get_vfs_name(args.vfs_path)

    print_configuration(args)

    if args.script_path:
        result = run_startup_script(
            args.script_path,
            vfs_name
        )

        if result == "exit":
            return

    run_repl(vfs_name)


if __name__ == "__main__":
    main()
